//! Host-side typed builders for linked-book swaps and maker page operations.
//! This module is excluded from the SBF target to preserve the pinned ELF limit.
use crate::ID;
use solana_program::{
    instruction::{AccountMeta, Instruction},
    pubkey::Pubkey,
};

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub enum Side {
    Ask,
    Bid,
}
impl Side {
    pub const fn tag(self) -> u8 {
        match self {
            Self::Ask => 0,
            Self::Bid => 1,
        }
    }
    const fn page_seed(self) -> &'static [u8] {
        match self {
            Self::Ask => b"ask-page",
            Self::Bid => b"bid-page",
        }
    }
    const fn owner_seed(self) -> &'static [u8] {
        match self {
            Self::Ask => b"ask-owner-page",
            Self::Bid => b"bid-owner-page",
        }
    }
}
#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub enum BuildError {
    InvalidPageCount,
    InvalidPageIndex,
    MissingMakerAccounts,
    InvalidAmount,
    DuplicateAccount,
    InvalidPrice,
}

#[derive(Clone, Copy, Debug)]
pub struct PageAccounts {
    pub page: Pubkey,
    pub owner: Pubkey,
}

/// The head owner sidecar may be a legacy non-PDA market-bound account.
/// Index > 0 must use its canonical PDA.
pub fn canonical_page(market: Pubkey, side: Side, index: u32) -> PageAccounts {
    let bytes = index.to_le_bytes();
    PageAccounts {
        page: Pubkey::find_program_address(&[side.page_seed(), market.as_ref(), &bytes], &ID).0,
        owner: Pubkey::find_program_address(&[side.owner_seed(), market.as_ref(), &bytes], &ID).0,
    }
}
pub fn maker_balance(market: Pubkey, owner: Pubkey) -> Pubkey {
    Pubkey::find_program_address(&[b"maker-balance", market.as_ref(), owner.as_ref()], &ID).0
}
pub fn custody(market: Pubkey) -> Pubkey {
    Pubkey::find_program_address(&[b"custody", market.as_ref()], &ID).0
}

fn ensure_unique(keys: &[Pubkey]) -> Result<(), BuildError> {
    for (i, k) in keys.iter().enumerate() {
        if keys[i + 1..].contains(k) {
            return Err(BuildError::DuplicateAccount);
        }
    }
    Ok(())
}
#[derive(Clone, Copy)]
pub struct TwoPageBook {
    pub market: Pubkey,
    pub head: PageAccounts,
    pub tail: PageAccounts,
    pub side: Side,
}
impl TwoPageBook {
    pub fn new(market: Pubkey, head_owner: Pubkey, side: Side) -> Self {
        let mut head = canonical_page(market, side, 0);
        head.owner = head_owner;
        Self {
            market,
            head,
            tail: canonical_page(market, side, 1),
            side,
        }
    }
    fn accounts(self, market_signer: bool) -> Result<Vec<AccountMeta>, BuildError> {
        ensure_unique(&[
            self.market,
            self.head.page,
            self.head.owner,
            self.tail.page,
            self.tail.owner,
        ])?;
        Ok(vec![
            AccountMeta::new(self.market, market_signer),
            AccountMeta::new(self.head.page, false),
            AccountMeta::new(self.head.owner, false),
            AccountMeta::new(self.tail.page, false),
            AccountMeta::new(self.tail.owner, false),
        ])
    }
    /// Opcode 29, signed by the market key. Only succeeds on an empty book.
    pub fn attach(self) -> Result<Instruction, BuildError> {
        self.transition(29)
    }
    /// Opcode 32, signed by the market key. Only succeeds when tail is empty.
    pub fn detach(self) -> Result<Instruction, BuildError> {
        self.transition(32)
    }
    fn transition(self, opcode: u8) -> Result<Instruction, BuildError> {
        Ok(Instruction {
            program_id: ID,
            accounts: self.accounts(true)?,
            data: vec![opcode, self.side.tag()],
        })
    }
    /// Opcode 30. Maker must sign, and market must have sufficient deposited free collateral.
    pub fn place(
        self,
        index: u8,
        maker: Pubkey,
        price_x64: u128,
        sqrt_price_x64: u128,
        base_qty: u64,
    ) -> Result<Instruction, BuildError> {
        if index > 1 {
            return Err(BuildError::InvalidPageIndex);
        }
        if base_qty == 0 || price_x64 == 0 || sqrt_price_x64 == 0 {
            return Err(BuildError::InvalidPrice);
        }
        let mut accounts = self.accounts(false)?;
        let balance = maker_balance(self.market, maker);
        ensure_unique(&[
            self.market,
            self.head.page,
            self.head.owner,
            self.tail.page,
            self.tail.owner,
            balance,
            maker,
        ])?;
        accounts.push(AccountMeta::new(balance, false));
        accounts.push(AccountMeta::new_readonly(maker, true));
        let mut data = vec![30, self.side.tag(), index];
        data.extend_from_slice(&price_x64.to_le_bytes());
        data.extend_from_slice(&sqrt_price_x64.to_le_bytes());
        data.extend_from_slice(&base_qty.to_le_bytes());
        Ok(Instruction {
            program_id: ID,
            accounts,
            data,
        })
    }
    /// Opcode 31. Maker must sign. The page index is explicit and must match the sequence owner.
    pub fn cancel(
        self,
        index: u8,
        maker: Pubkey,
        sequence: u64,
    ) -> Result<Instruction, BuildError> {
        if index > 1 {
            return Err(BuildError::InvalidPageIndex);
        }
        if sequence == 0 {
            return Err(BuildError::InvalidAmount);
        }
        let mut accounts = self.accounts(false)?;
        let balance = maker_balance(self.market, maker);
        ensure_unique(&[
            self.market,
            self.head.page,
            self.head.owner,
            self.tail.page,
            self.tail.owner,
            balance,
            maker,
        ])?;
        accounts.push(AccountMeta::new(balance, false));
        accounts.push(AccountMeta::new_readonly(maker, true));
        let mut data = vec![31, self.side.tag(), index];
        data.extend_from_slice(&sequence.to_le_bytes());
        Ok(Instruction {
            program_id: ID,
            accounts,
            data,
        })
    }
    /// Opcode 27: close an empty detached page 1, returning rent to the market.
    pub fn close_detached(self) -> Result<Instruction, BuildError> {
        ensure_unique(&[self.market, self.tail.page, self.tail.owner])?;
        let mut data = vec![27, self.side.tag()];
        data.extend_from_slice(&1u32.to_le_bytes());
        Ok(Instruction {
            program_id: ID,
            accounts: vec![
                AccountMeta::new(self.market, false),
                AccountMeta::new(self.tail.page, false),
                AccountMeta::new(self.tail.owner, false),
            ],
            data,
        })
    }
    /// Opcode 28: recreate the detached page-1 PDA pair, payer-funded.
    pub fn recreate_detached(
        self,
        payer: Pubkey,
        system_program: Pubkey,
    ) -> Result<Instruction, BuildError> {
        ensure_unique(&[
            self.market,
            self.head.page,
            self.head.owner,
            self.tail.page,
            self.tail.owner,
            payer,
            system_program,
        ])?;
        let mut accounts = self.accounts(false)?;
        accounts.push(AccountMeta::new(payer, true));
        accounts.push(AccountMeta::new_readonly(system_program, false));
        Ok(Instruction {
            program_id: ID,
            accounts,
            data: vec![28, self.side.tag()],
        })
    }
}

/// Opcode 22: read-only on-chain validation of linked page and sidecar chains.
pub fn linked_book_preflight(market: Pubkey, side: Side, pages: &[PageAccounts]) -> Result<Instruction, BuildError> {
    if pages.is_empty() || pages.len() > 8 { return Err(BuildError::InvalidPageCount); }
    let mut keys = vec![market];
    for p in pages { keys.push(p.page); keys.push(p.owner); }
    ensure_unique(&keys)?;
    let mut accounts = vec![AccountMeta::new_readonly(market,false)];
    for p in pages { accounts.push(AccountMeta::new_readonly(p.page,false)); accounts.push(AccountMeta::new_readonly(p.owner,false)); }
    Ok(Instruction{program_id:ID,accounts,data:vec![22,side.tag(),pages.len() as u8]})
}

/// Accounts for a bounded exact-in/out SPL Token swap. For buys the taker and
/// vault order is quote then base; for sells, base then quote.
pub struct LinkedSwap {
    pub market: Pubkey,
    pub custody: Pubkey,
    pub taker: Pubkey,
    pub taker_base: Pubkey,
    pub taker_quote: Pubkey,
    pub base_vault: Pubkey,
    pub quote_vault: Pubkey,
    pub vault_authority: Pubkey,
    pub token_program: Pubkey,
    pub pages: Vec<PageAccounts>,
    pub maker_balances: Vec<Pubkey>,
}
impl LinkedSwap {
    /// Opcode 23/24 for buy and 25/26 for sell. The caller supplies current
    /// canonical book pages and exact maker balance PDA accounts in chain order.
    pub fn instruction(
        &self,
        side: Side,
        exact_out: bool,
        amount: u64,
        limit: u64,
    ) -> Result<Instruction, BuildError> {
        if self.pages.is_empty() || self.pages.len() > 8 {
            return Err(BuildError::InvalidPageCount);
        }
        if self.maker_balances.is_empty() || self.maker_balances.len() > 8 {
            return Err(BuildError::MissingMakerAccounts);
        }
        if amount == 0 || limit == 0 {
            return Err(BuildError::InvalidAmount);
        }
        let buy = side == Side::Ask;
        let (taker_in, taker_out, vault_in, vault_out) = if buy {
            (
                self.taker_quote,
                self.taker_base,
                self.quote_vault,
                self.base_vault,
            )
        } else {
            (
                self.taker_base,
                self.taker_quote,
                self.base_vault,
                self.quote_vault,
            )
        };
        let mut keys = vec![
            self.market,
            self.custody,
            self.taker,
            taker_in,
            taker_out,
            vault_in,
            vault_out,
            self.vault_authority,
            self.token_program,
        ];
        for p in &self.pages {
            keys.push(p.page);
            keys.push(p.owner);
        }
        keys.extend(self.maker_balances.iter().copied());
        ensure_unique(&keys)?;
        let mut metas = vec![
            AccountMeta::new(self.market, false),
            AccountMeta::new(self.custody, false),
            AccountMeta::new_readonly(self.taker, true),
            AccountMeta::new(taker_in, false),
            AccountMeta::new(taker_out, false),
            AccountMeta::new(vault_in, false),
            AccountMeta::new(vault_out, false),
            AccountMeta::new_readonly(self.vault_authority, false),
            AccountMeta::new_readonly(self.token_program, false),
        ];
        for p in &self.pages {
            metas.push(AccountMeta::new(p.page, false));
            metas.push(AccountMeta::new(p.owner, false));
        }
        for b in &self.maker_balances {
            metas.push(AccountMeta::new(*b, false));
        }
        let opcode = match (buy, exact_out) {
            (true, false) => 23,
            (true, true) => 24,
            (false, false) => 25,
            (false, true) => 26,
        };
        let mut data = vec![opcode, self.pages.len() as u8];
        data.extend_from_slice(&amount.to_le_bytes());
        data.extend_from_slice(&limit.to_le_bytes());
        Ok(Instruction {
            program_id: ID,
            accounts: metas,
            data,
        })
    }
}
#[cfg(test)]
mod tests {
    use super::*;
    #[test]
    fn pda_derivation_is_market_and_side_bound() {
        let m = Pubkey::new_unique();
        assert_ne!(
            canonical_page(m, Side::Ask, 1).page,
            canonical_page(m, Side::Bid, 1).page
        );
        assert_ne!(
            canonical_page(m, Side::Ask, 1).owner,
            canonical_page(m, Side::Ask, 0).owner
        );
        assert_ne!(
            maker_balance(m, Pubkey::new_unique()),
            maker_balance(m, Pubkey::new_unique())
        );
    }
    #[test]
    fn full_book_lifecycle_encodes_exact_abi_and_signers() {
        for side in [Side::Ask, Side::Bid] {
            let market = Pubkey::new_unique();
            let owner = Pubkey::new_unique();
            let maker = Pubkey::new_unique();
            let b = TwoPageBook::new(market, owner, side);
            for (opcode, ix) in [(29, b.attach().unwrap()), (32, b.detach().unwrap())] {
                assert_eq!(ix.data, vec![opcode, side.tag()]);
                assert_eq!(ix.accounts.len(), 5);
                assert!(ix.accounts[0].is_signer);
            }
            let ix = b.place(1, maker, 1 << 64, 1 << 64, 25).unwrap();
            assert_eq!(ix.data.len(), 43);
            assert_eq!(ix.accounts.len(), 7);
            assert!(ix.accounts[6].is_signer);
            assert_eq!(&ix.data[3..19], &(1u128 << 64).to_le_bytes());
            assert_eq!(&ix.data[35..43], &25u64.to_le_bytes());
            let ix = b.cancel(1, maker, 9).unwrap();
            assert_eq!(ix.data.len(), 11);
            assert_eq!(&ix.data[3..11], &9u64.to_le_bytes());
            assert_eq!(b.close_detached().unwrap().data.len(), 6);
            assert_eq!(
                b.recreate_detached(Pubkey::new_unique(), Pubkey::new_unique())
                    .unwrap()
                    .accounts
                    .len(),
                7
            );
            assert_eq!(
                b.place(2, maker, 1, 1, 1).err(),
                Some(BuildError::InvalidPageIndex)
            );
            assert_eq!(b.cancel(0, maker, 0).err(), Some(BuildError::InvalidAmount));
        }
    }
    #[test]
    fn swap_buy_sell_exact_in_out_account_order_and_rejection() {
        let market = Pubkey::new_unique();
        let keys: Vec<_> = (0..10).map(|_| Pubkey::new_unique()).collect();
        let mut s = LinkedSwap {
            market,
            custody: keys[0],
            taker: keys[1],
            taker_base: keys[2],
            taker_quote: keys[3],
            base_vault: keys[4],
            quote_vault: keys[5],
            vault_authority: keys[6],
            token_program: keys[7],
            pages: vec![PageAccounts {
                page: keys[8],
                owner: keys[9],
            }],
            maker_balances: vec![Pubkey::new_unique()],
        };
        for (side, exact_out, opcode) in [
            (Side::Ask, false, 23),
            (Side::Ask, true, 24),
            (Side::Bid, false, 25),
            (Side::Bid, true, 26),
        ] {
            let ix = s.instruction(side, exact_out, 30, 40).unwrap();
            assert_eq!(ix.data[0], opcode);
            assert_eq!(ix.data[1], 1);
            assert_eq!(ix.data.len(), 18);
            assert_eq!(ix.accounts.len(), 12);
            assert_eq!(
                ix.accounts[3].pubkey,
                if side == Side::Ask {
                    s.taker_quote
                } else {
                    s.taker_base
                }
            );
            assert_eq!(
                ix.accounts[4].pubkey,
                if side == Side::Ask {
                    s.taker_base
                } else {
                    s.taker_quote
                }
            );
            assert!(ix.accounts[2].is_signer);
        }
        assert_eq!(
            s.instruction(Side::Ask, false, 0, 10).err(),
            Some(BuildError::InvalidAmount)
        );
        let check = linked_book_preflight(market, Side::Ask, &s.pages).unwrap();
        assert_eq!(check.data, vec![22, 0, 1]);
        assert_eq!(check.accounts.len(),3);
        assert!(check.accounts.iter().all(|m| !m.is_writable && !m.is_signer));
        s.maker_balances.extend((0..8).map(|_| Pubkey::new_unique()));
        assert_eq!(s.instruction(Side::Ask,false,1,1).err(),Some(BuildError::MissingMakerAccounts));
        s.maker_balances.truncate(1);
        s.pages.clear();
        assert_eq!(
            s.instruction(Side::Ask, false, 1, 10).err(),
            Some(BuildError::InvalidPageCount)
        );
    }
}
