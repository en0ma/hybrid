//! Offline audit of two-page market state and safe next-step planning.
//! All decisions require a fresh snapshot; transaction execution must revalidate.
use crate::client::{canonical_page, PageAccounts, Side, TwoPageBook};
use solana_program::pubkey::Pubkey;
use hybrid_state::{AskOwnerPage, AskPage, BidPage, MarketHeader, PageLinks};

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub enum AuditError {
    InvalidMarket, BadPage, BadSidecar, WrongAddress, WrongOwner,
    InconsistentLinks, InconsistentCount, DuplicateSequence, Unsorted,
}
#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub enum NextAction {
    /// Safe to attach an empty detached page, if a market signer exists.
    AttachEmpty,
    /// A currently linked tail is empty; can detach with market signature.
    DetachEmpty,
    /// Tail is detached and empty; can close and reclaim rent into the market.
    CloseDetached,
    /// Tail is linked and populated; orders can be cancelled or filled.
    CancelOrFill,
    /// Head has orders; may place orders on head, but cannot safely attach tail.
    HeadOrders,
    /// No operational transition is needed.
    None,
}
#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub struct AuditReport {
    pub side: Side,
    pub head_orders: usize,
    pub tail_orders: usize,
    pub next_action: NextAction,
    pub is_linked: bool,
    pub total_orders: usize,
}
/// Copies of a decoded account view. Ownership and address are retained for
/// independent authentication; no RPC is performed by the auditing module.
pub struct AccountView<'a> {
    pub key: Pubkey,
    pub owner: Pubkey,
    pub data: &'a [u8],
}
pub struct TwoPageSnapshot<'a> {
    pub market: AccountView<'a>,
    pub head: AccountView<'a>,
    pub head_owner: AccountView<'a>,
    pub tail: AccountView<'a>,
    pub tail_owner: AccountView<'a>,
    pub side: Side,
}

fn sidecar_tag(key: Pubkey) -> [u8; 16] {
    let mut tag = [0; 16];
    tag.copy_from_slice(&key.to_bytes()[..16]);
    tag
}
impl<'a> TwoPageSnapshot<'a> {
    pub fn audit(&self) -> Result<AuditReport, AuditError> {
        let market_key = self.market.key;
        let expected_head = canonical_page(market_key, self.side, 0).page;
        let expected_tail = canonical_page(market_key, self.side, 1);
        if self.head.key != expected_head || self.tail.key != expected_tail.page
            || self.tail_owner.key != expected_tail.owner {
            return Err(AuditError::WrongAddress);
        }
        let accounts = [&self.market, &self.head, &self.head_owner, &self.tail, &self.tail_owner];
        if accounts.iter().any(|a| a.owner != crate::ID) {
            return Err(AuditError::WrongOwner);
        }
        for (i,a) in accounts.iter().enumerate() {
            if accounts[i+1..].iter().any(|b| a.key == b.key) {
                return Err(AuditError::WrongAddress);
            }
        }
        let header = MarketHeader::decode_from(self.market.data)
            .map_err(|_| AuditError::InvalidMarket)?;
        if !header.collateralized_active() { return Err(AuditError::InvalidMarket); }
        match self.side {
            Side::Ask if header.reserved2 != self.head_owner.key.to_bytes() => {
                return Err(AuditError::WrongAddress);
            }
            Side::Bid if header.bid_owner_tag() != sidecar_tag(self.head_owner.key) => {
                return Err(AuditError::WrongAddress);
            }
            _ => {}
        }
        let howner = AskOwnerPage::decode_from(self.head_owner.data)
            .map_err(|_| AuditError::BadSidecar)?;
        let towner = AskOwnerPage::decode_from(self.tail_owner.data)
            .map_err(|_| AuditError::BadSidecar)?;
        let (head_len,tail_len,head_links,tail_links,sequences,ordered) = match self.side {
            Side::Ask => {
                let h=AskPage::decode_from(self.head.data).map_err(|_| AuditError::BadPage)?;
                let t=AskPage::decode_from(self.tail.data).map_err(|_| AuditError::BadPage)?;
                howner.validate_parallel(&h).map_err(|_| AuditError::BadSidecar)?;
                towner.validate_parallel(&t).map_err(|_| AuditError::BadSidecar)?;
                let ordered=h.validate_order().is_ok() && t.validate_order().is_ok();
                let sequences=h.as_slice().iter().chain(t.as_slice()).map(|x|x.sequence).collect::<Vec<_>>();
                (h.len(),t.len(),h.links(),t.links(),sequences,ordered && (h.links().next_page!=Some(1) || h.validate_next(&t).is_ok()))
            }
            Side::Bid => {
                let h=BidPage::decode_from(self.head.data).map_err(|_| AuditError::BadPage)?;
                let t=BidPage::decode_from(self.tail.data).map_err(|_| AuditError::BadPage)?;
                if howner.len()!=h.len() || towner.len()!=t.len()
                    || howner.links()!=h.links() || towner.links()!=t.links()
                    || howner.as_slice().iter().chain(towner.as_slice()).any(|x|*x==[0u8;32]) {
                    return Err(AuditError::BadSidecar);
                }
                let ordered=h.validate_order().is_ok() && t.validate_order().is_ok();
                let sequences=h.as_slice().iter().chain(t.as_slice()).map(|x|x.sequence).collect::<Vec<_>>();
                (h.len(),t.len(),h.links(),t.links(),sequences,ordered && (h.links().next_page!=Some(1) || h.validate_next(&t).is_ok()))
            }
        };
        if !ordered {return Err(AuditError::Unsorted);}
        for (i,seq) in sequences.iter().enumerate() {
            if *seq==0 || sequences[i+1..].contains(seq) {return Err(AuditError::DuplicateSequence);}
        }
        let linked=head_links==PageLinks::new(0,None,Some(1))
            && tail_links==PageLinks::new(1,Some(0),None);
        let detached=head_links==PageLinks::new(0,None,None)
            && tail_links==PageLinks::new(1,None,None);
        if !linked && !detached {return Err(AuditError::InconsistentLinks);}
        if (match self.side {Side::Ask=>header.ask_count,Side::Bid=>header.bid_count()}) as usize
            != head_len+tail_len {return Err(AuditError::InconsistentCount);}
        if detached && tail_len!=0 {return Err(AuditError::InconsistentLinks);}
        let next_action=if linked {
            if tail_len==0 {NextAction::DetachEmpty} else {NextAction::CancelOrFill}
        } else if head_len==0 {NextAction::AttachEmpty}
        else {NextAction::HeadOrders};
        Ok(AuditReport{
            side:self.side,head_orders:head_len,tail_orders:tail_len,
            next_action,is_linked:linked,total_orders:head_len+tail_len,
        })
    }
    pub fn builder(&self) -> Result<TwoPageBook,AuditError> {
        self.audit()?;
        Ok(TwoPageBook::new(self.market.key,self.head_owner.key,self.side))
    }
    pub fn expected_addresses(&self) -> (PageAccounts,PageAccounts) {
        let mut head=canonical_page(self.market.key,self.side,0);
        head.owner=self.head_owner.key;
        (head,canonical_page(self.market.key,self.side,1))
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use hybrid_engine::Q64;
    use hybrid_state::{AskEntry, MARKET_FLAG_COLLATERALIZED_ACTIVE,
        MARKET_HEADER_BYTES, ASK_PAGE_BYTES, ASK_OWNER_PAGE_BYTES, BID_PAGE_BYTES};
    fn valid(side: Side, linked: bool, count: usize) -> (Pubkey,Vec<u8>,Vec<u8>,Vec<u8>,Vec<u8>,Vec<u8>,Pubkey) {
        let market=Pubkey::new_unique();
        let head_owner=Pubkey::new_unique();
        let mut header=MarketHeader::new(1,Q64,1_000_000,1,1);
        header.flags |= MARKET_FLAG_COLLATERALIZED_ACTIVE;
        if side==Side::Ask {header.ask_count=count as u32;header.reserved2=head_owner.to_bytes();}
        else {header.set_bid_count(count as u32);header.set_bid_owner_tag(sidecar_tag(head_owner));}
        let mut mh=vec![0;MARKET_HEADER_BYTES];header.encode_into(&mut mh).unwrap();
        let mut a=vec![0;if side==Side::Ask{ASK_PAGE_BYTES}else{BID_PAGE_BYTES}];
        let mut b=a.clone();
        let mut ho=AskOwnerPage::default();let mut to=AskOwnerPage::default();
        let hlinks=PageLinks::new(0,None,linked.then_some(1));
        let tlinks=PageLinks::new(1,linked.then_some(0),None);
        ho.set_links(hlinks);to.set_links(tlinks);
        let mut oh=vec![0;ASK_OWNER_PAGE_BYTES];let mut ot=oh.clone();
        if side==Side::Ask {
            let mut h=AskPage::default();let mut t=AskPage::default();
            h.set_links(hlinks);t.set_links(tlinks);
            if count>0 {h.insert(AskEntry{price_x64:Q64,sqrt_price_x64:Q64,base_qty:10,sequence:1}).unwrap();
                ho.len=1;ho.owners[0]=Pubkey::new_unique().to_bytes();}
            h.encode_into(&mut a).unwrap();t.encode_into(&mut b).unwrap();
        } else {
            let mut h=BidPage::default();let mut t=BidPage::default();
            h.set_links(hlinks);t.set_links(tlinks);
            if count>0 {h.insert(AskEntry{price_x64:Q64,sqrt_price_x64:Q64,base_qty:10,sequence:1}).unwrap();
                ho.len=1;ho.owners[0]=Pubkey::new_unique().to_bytes();}
            h.encode_into(&mut a).unwrap();t.encode_into(&mut b).unwrap();
        }
        ho.encode_into(&mut oh).unwrap();to.encode_into(&mut ot).unwrap();
        (market,mh,a,b,oh,ot,head_owner)
    }
    #[test]
    fn audit_detects_safe_link_transitions_and_tampering() {
        for side in [Side::Ask,Side::Bid] {
            for (linked,count,expected) in [
                (false,0,NextAction::AttachEmpty),
                (true,0,NextAction::DetachEmpty),
                (true,1,NextAction::DetachEmpty),
                (false,1,NextAction::HeadOrders),
            ]{
                let (market,mut mh,h,t,ho,to,owner)=valid(side,linked,count);
                let hp=canonical_page(market,side,0);
                let tp=canonical_page(market,side,1);
                let snap=|market_data:&[u8]|TwoPageSnapshot{
                    market:AccountView{key:market,owner:crate::ID,data:market_data},
                    head:AccountView{key:hp.page,owner:crate::ID,data:&h},
                    head_owner:AccountView{key:owner,owner:crate::ID,data:&ho},
                    tail:AccountView{key:tp.page,owner:crate::ID,data:&t},
                    tail_owner:AccountView{key:tp.owner,owner:crate::ID,data:&to},
                    side,
                };
                let result=snap(&mh).audit().unwrap();
                assert_eq!(result.next_action,expected);
                assert_eq!(result.total_orders,count);
                assert_eq!(snap(&mh).builder().unwrap().tail.page,tp.page);
                mh[12..16].copy_from_slice(&999u32.to_le_bytes());
                if side==Side::Ask {
                    assert_eq!(snap(&mh).audit(),Err(AuditError::InconsistentCount));
                }
            }
        }
    }
}
